    <script>
        const form = document.getElementById('customerForm');
        const resultDiv = document.getElementById('result');
        const loadingDiv = document.getElementById('loading');

        // Demo data
        document.getElementById('demoBtn').onclick = () => {
            form.age_group.value = '18-35';
            form.subscription_duration.value = '6-12 months';
            form.subscription_plan.value = 'Standard';
            form.monthly_spend.value = '₹200 - ₹500';
            form.usage_frequency.value = 'Often';
            form.value_perception.value = 'Average';
            form.nps_score.value = '7';
            form.content_satisfaction.value = '4';
            form.churn_likelihood.value = 'Not sure';
            form.switching_behavior.value = 'No changes';
            form.account_sharing.value = 'No, only I use it';
            form.competitor_awareness.value = 'No, satisfied with current';
            form.bundling_preference.value = 'Maybe, depends on price';
            form.payment_method.value = 'Monthly auto-renewal';
            form.primary_platform.value = 'Netflix';
        };

        // Clear form
        document.getElementById('clearBtn').onclick = () => form.reset();

        // Submit form
        form.onsubmit = async (e) => {
            e.preventDefault();
            
            const formData = new FormData(form);
            const data = {};
            for (let [key, value] of formData.entries()) {
                if (value) data[key] = value;
            }
            
            // Convert numbers
            if (data.nps_score) data.nps_score = parseInt(data.nps_score);
            if (data.content_satisfaction) data.content_satisfaction = parseInt(data.content_satisfaction);
            
            console.log('Sending:', data);
            
            loadingDiv.style.display = 'block';
            resultDiv.style.display = 'none';
            
            try {
                const response = await fetch('http://localhost:5001/api/data/customer', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(data)
                });
                
                const result = await response.json();
                console.log('Result:', result);
                
                if (result.success) {
                    const prob = (result.prediction?.churn_probability * 100).toFixed(1);
                    resultDiv.innerHTML = `
                        <h3>✅ Customer Saved! (ID: ${result.data.id})</h3>
                        <p><strong>Churn Probability:</strong> ${prob}%</p>
                        <p><strong>Risk Level:</strong> ${result.prediction?.risk_level || 'Medium'}</p>
                        <p><strong>Algorithm Used:</strong> ${data.algorithm === 'random_forest' ? 'Random Forest' : 'Logistic Regression'}</p>
                        <p><strong>Recommendation:</strong> ${result.prediction?.recommendations?.[0] || 'Monitor customer behavior'}</p>
                    `;
                    resultDiv.className = 'result success';
                } else {
                    resultDiv.innerHTML = `<p class="error">Error: ${result.message}</p>`;
                    resultDiv.className = 'result error';
                }
                resultDiv.style.display = 'block';
                
            } catch (error) {
                resultDiv.innerHTML = `<p class="error">Connection error: ${error.message}</p>`;
                resultDiv.className = 'result error';
                resultDiv.style.display = 'block';
            } finally {
                loadingDiv.style.display = 'none';
            }
        };
    </script>
